def _safe_trigger(trigger_name: str, expected_state: str = None) -> bool:
    """
    Fire a state machine trigger ONLY if the current state is valid.
    Prevents MachineError race condition crashes.
    """
    current = sm.state
    if expected_state and current != expected_state:
        logger.warning(
            f"[SM] Blocked '{trigger_name}': expected '{expected_state}', got '{current}'"
        )
        return False
    try:
        getattr(sm, trigger_name)()
        logger.info(f"[SM] {trigger_name}() → {sm.state}")
        return True
    except Exception as e:
        logger.warning(f"[SM] '{trigger_name}' failed in state '{current}': {e}")
        return False

def generate_receipt(session: dict) -> dict:
    """Generate official SEVA AI application receipt with tracking ID."""
    now = datetime.datetime.now()
    # Unique Application ID: SEVA-YYYYMMDD-XXXX

    return {
        "application_id":  app_id,
        "timestamp":       now.strftime("%d %B %Y, %I:%M %p"),
        "scheme_name":     session.get("scheme", "Sanjay Gandhi Niradhar Anudan Yojana"),
        "applicant_name":  session["fields"].get("name", "Verified Citizen"),

async def push_ui(event: str, data: dict = {}):
    payload = json.dumps({"event": event, **data})
    dead = []
    for ws in ui_clients:
        try:
            await ws.send_text(payload)
        except Exception:
            dead.append(ws)
    for d in dead:
        if d in ui_clients:
            ui_clients.remove(d)

async def run_real_session():
    lang = _session["language"]

    _safe_trigger("citizen_detected", "IDLE")
    await _say_push("greeting", "INTENT_DISCOVERY")

    # ASR — run in thread to avoid blocking event loop
    try:
        intent_text = await asyncio.wait_for(
            asyncio.to_thread(asr.listen_and_transcribe, lang_code=lang, duration=5.0),
            timeout=20.0,   # Hard timeout — prevents forever hang
        )
    except asyncio.TimeoutError:
        logger.warning("[HOST] ASR timed out — continuing without voice input.")
        intent_text = "(voice input unavailable)"
    except Exception as e:
        logger.error(f"[HOST] ASR error: {e}")
        intent_text = ""

    await push_ui("INTENT_CAPTURED", {"text": intent_text, "message": intent_text})

    # FIX: Guard the transition — only fire if in correct state
    if not _safe_trigger("intent_confirmed", "INTENT_DISCOVERY"):
        logger.warning("[HOST] Aborting session — bad state after ASR.")
        return

    await _say_push("scan_doc", "DOCUMENT_SCANNING")

    # Camera capture — in thread
    try:
        frame = await asyncio.wait_for(
            asyncio.to_thread(_capture_document),
            timeout=15.0,
        )
    except asyncio.TimeoutError:
        logger.error("[HOST] Camera timed out.")
        frame = None
    except Exception as e:
        logger.error(f"[HOST] Camera error: {e}")
        frame = None

    if frame is None:
        await _say_push("error", "ERROR", {"detail": "Camera capture failed."})
        _safe_trigger("error_occurred")
        return

    # OCR — in thread
    try:
        extracted = await asyncio.to_thread(ocr.extract, frame, lang)
        await asyncio.to_thread(ocr.unload, lang)
    except Exception as e:
        logger.error(f"[HOST] OCR error: {e}")
        extracted = {"fields": {}}

    fields = extracted["fields"]
    _session["fields"] = fields
    if not _safe_trigger("docs_verified", "DOCUMENT_SCANNING"):
        return

    await push_ui("OCR_DONE", {"fields": fields, "message": "Document scanned."})
    await _say_push("confirm", "PORTAL_AUTOMATION")

    # Build browser agent task
    task = (
        f"Go to mahaonline.gov.in and apply for Sanjay Gandhi Niradhar Anudan Yojana. "
        f"Fill the form with: Name={fields.get('name','')}, "
        f"DOB={fields.get('dob','')}, Gender={fields.get('gender','Female')}. "
        f"STOP — do NOT type anything into OTP, password, or PIN fields."
    )

    await _say_push("filling", "PORTAL_AUTOMATION", {"task": task})

    vm_bridge.set_handlers(_on_checkpoint, _on_done, _on_error)
    try:
        await vm_bridge.send_task(task)
        await vm_bridge.listen()
    except Exception as e:
        logger.error(f"[HOST] VM failed: {e}")
        await _say_push("error", "ERROR", {"detail": str(e)})
        _safe_trigger("error_occurred")

def _base_env(demo: bool) -> dict:
    """Build the environment dict for child processes with all required fixes."""
    env = os.environ.copy()
    # FIX: loopback proxy bypass (prevents Gradio → FastAPI connection issues)
    env["NO_PROXY"]                      = "localhost,127.0.0.1,::1"
    env["no_proxy"]                      = "localhost,127.0.0.1,::1"
    # FIX: Windows symlink privilege (WinError 1314)

    env["HF_HUB_ENABLE_HF_TRANSFER"]       = "0"
    env["TRANSFORMERS_OFFLINE"]             = "0"

        env["DEMO_MODE"] = "true"
    return env


def start_fastapi(demo: bool) -> subprocess.Popen:
    env = _base_env(demo)
    cmd = [sys.executable, "-m", "uvicorn", "main:app",
           "--host", "0.0.0.0", "--port", str(API_PORT), "--log-level", "info"]
    logger.info(f"[FastAPI] Starting on :{API_PORT} {'(DEMO)' if demo else ''}")
    return subprocess.Popen(cmd, cwd=str(HERE), env=env)


def start_gradio() -> subprocess.Popen:
    env = _base_env(False)
    env["GRADIO_PORT"] = str(UI_PORT)
    cmd = [sys.executable, str(HERE / "ui" / "gradio_app.py")]
    logger.info(f"[Gradio] Starting on :{UI_PORT}")

"""
vdak-host/state_machine.py
Hierarchical Finite State Machine for V-DAK kiosk session lifecycle.
States: IDLE → INTENT_DISCOVERY → DOCUMENT_SCANNING →
        PORTAL_AUTOMATION → HUMAN_CHECKPOINT → SUBMISSION → PURGE
"""

import logging
from transitions import Machine

logger = logging.getLogger("vdak.statemachine")


STATES = [
    "IDLE",
    "INTENT_DISCOVERY",
    "DOCUMENT_SCANNING",
    "PORTAL_AUTOMATION",
    "HUMAN_CHECKPOINT",
    "SUBMISSION",
    "PURGE",
    "ERROR",
]

TRANSITIONS = [
    # Citizen walks up → greeting
    {"trigger": "citizen_detected",   "source": "IDLE",              "dest": "INTENT_DISCOVERY"},
    # Intent understood → scan documents
    {"trigger": "intent_confirmed",   "source": "INTENT_DISCOVERY",  "dest": "DOCUMENT_SCANNING"},
    # OCR done and verified → launch browser agent on VM
    {"trigger": "docs_verified",      "source": "DOCUMENT_SCANNING", "dest": "PORTAL_AUTOMATION"},
    # OTP field detected by VM agent
    {"trigger": "checkpoint_hit",     "source": "PORTAL_AUTOMATION", "dest": "HUMAN_CHECKPOINT"},
    # Citizen typed OTP → agent resumes
    {"trigger": "otp_submitted",      "source": "HUMAN_CHECKPOINT",  "dest": "PORTAL_AUTOMATION"},
    # Form successfully submitted
    {"trigger": "submission_done",    "source": "PORTAL_AUTOMATION", "dest": "SUBMISSION"},
    # Print receipt → wipe session
    {"trigger": "receipt_printed",    "source": "SUBMISSION",        "dest": "PURGE"},
    # After purge → back to idle
    {"trigger": "session_purged",     "source": "PURGE",             "dest": "IDLE"},
    # Any error → ERROR state
    {"trigger": "error_occurred",     "source": "*",                 "dest": "ERROR"},
    # Recover from error
    {"trigger": "reset",              "source": "ERROR",             "dest": "IDLE"},
    # User cancels / timeout
    {"trigger": "user_cancelled",     "source": "*",                 "dest": "PURGE"},
]


class KioskStateMachine:
    """V-DAK session state machine with logging on every transition."""

    def __init__(self, on_state_change=None):
        """
        on_state_change: optional async callback(new_state: str)
        called after every transition — used to update the React UI.
        """
        self._on_state_change = on_state_change

        self.machine = Machine(
            model=self,
            states=STATES,
            transitions=TRANSITIONS,
            initial="IDLE",
            after_state_change="_after_transition",
        )

    def _after_transition(self):
        logger.info(f"[STATE] → {self.state}")
        if self._on_state_change:
            import asyncio
            try:
                loop = asyncio.get_event_loop()
                if loop.is_running():
                    asyncio.ensure_future(self._on_state_change(self.state))
            except RuntimeError:
                pass

    @property
    def current(self) -> str:
        return self.state

    def is_idle(self) -> bool:
        return self.state == "IDLE"

    def is_waiting_for_otp(self) -> bool:
        return self.state == "HUMAN_CHECKPOINT"

"""
vdak-host/bridge/ws_client.py
WebSocket client on the host side.
Sends tasks to the VM browser agent and handles events back.
"""

import asyncio
import json
import logging
from typing import Callable, Optional

import websockets

logger = logging.getLogger("vdak.bridge")


class VMBridge:
    """
    Persistent WebSocket connection to the VM browser agent.
    Handles: RUN_TASK → receives CHECKPOINT / DONE / ERROR events.
    """

    def __init__(self, vm_ws_url: str):
        self.vm_ws_url = vm_ws_url
        self._ws = None
        self._on_checkpoint: Optional[Callable] = None
        self._on_done: Optional[Callable]       = None
        self._on_error: Optional[Callable]      = None

    def set_handlers(
        self,
        on_checkpoint: Callable,
        on_done: Callable,
        on_error: Callable,
    ):
        self._on_checkpoint = on_checkpoint
        self._on_done       = on_done
        self._on_error      = on_error

    async def connect(self):
        """Connect to VM WebSocket server."""
        logger.info(f"[BRIDGE] Connecting to VM at {self.vm_ws_url}...")
        self._ws = await websockets.connect(self.vm_ws_url)
        logger.info("[BRIDGE] Connected to VM.")

    async def send_task(self, task_description: str):
        """Send a RUN_TASK command to the VM browser agent."""
        if not self._ws:
            raise RuntimeError("Not connected to VM. Call connect() first.")
        payload = json.dumps({"type": "RUN_TASK", "task": task_description})
        await self._ws.send(payload)
        logger.info(f"[BRIDGE] Task sent to VM: {task_description[:80]}...")

    async def send_resume(self):
        """Tell VM that citizen has submitted OTP — agent can continue."""
        if not self._ws:
            raise RuntimeError("Not connected to VM.")
        await self._ws.send(json.dumps({"type": "RESUME_OTP"}))
        logger.info("[BRIDGE] RESUME_OTP sent to VM.")

    async def listen(self):
        """
        Event loop — listens for VM events and dispatches to handlers.
        Runs until connection closes.
        """
        async for raw in self._ws:
            event = json.loads(raw)
            etype = event.get("event")
            logger.info(f"[BRIDGE] VM event: {etype}")

            if etype == "HUMAN_CHECKPOINT" and self._on_checkpoint:
                await self._on_checkpoint()

            elif etype == "DONE" and self._on_done:
                await self._on_done(event.get("result", ""))

            elif etype == "ERROR" and self._on_error:

_state = {
    "event":          "IDLE",

    "fields":         {},

_lock = threading.Lock()


def _set(k, v):
    with _lock:
        _state[k] = v

def _get(k, d=None):
    with _lock:
        return _state.get(k, d)

def _log(msg):
    ts = time.strftime("%H:%M:%S")
    with _lock:
        _state["log"].append(f"[{ts}] {msg}")
        if len(_state["log"]) > 40:
            _state["log"] = _state["log"][-40:]


# ─────────────────────────────────────────────────────────────────────────────
#  WebSocket Listener (background daemon thread)
# ─────────────────────────────────────────────────────────────────────────────

def _ws_listener():
    import websocket as wsl

    def on_message(ws, raw):
        try:
            msg   = json.loads(raw)
            event = msg.get("event", "")
            with _lock:
                _state["event"]   = event
                _state["message"] = msg.get("message", msg.get("result", _state["message"]))
                if "fields" in msg and msg["fields"]:
                    _state["fields"] = msg["fields"]
                if "receipt" in msg and msg["receipt"]:
                    _state["receipt"] = msg["receipt"]

                # Decode live browser frame
                if event == "BROWSER_FRAME" and "data" in msg:
                    try:
                        img_bytes = base64.b64decode(msg["data"])
                        _state["browser_frame"] = Image.open(io.BytesIO(img_bytes))
                    except Exception:
                        pass
                elif event in ("IDLE", "PURGE"):
                    _state["browser_frame"] = None
                    _state["fields"]        = {}
                    _state["receipt"]       = None

            if event != "BROWSER_FRAME":
                _log(f"[{event}] {_state['message'][:60]}")
        except Exception as e:
            _log(f"[WS ERR] {e}")

    def on_open(ws):
        _set("connected", True)
        _log("✅ Connected to SEVA AI backend.")

    def on_close(ws, *_):
        _set("connected", False)
        _log("⚠️  Backend offline. Start FastAPI then refresh.")

    def on_error(ws, e):
        _log(f"[WS] {e}")

    while True:
        try:
            app = wsl.WebSocketApp(
                WS_URL,
                on_message=on_message, on_open=on_open,
                on_close=on_close,     on_error=on_error,
            )
            app.run_forever(ping_interval=20)
        except Exception as e:
            _log(f"[WS] Reconnect: {e}")
        time.sleep(3)

threading.Thread(target=_ws_listener, daemon=True).start()

"""
vdak-host/utils/vm_discover.py
Auto-discovers the VM's IP address on the local subnet (192.168.31.x).
Scans for an open port 8765 (V-DAK VM WebSocket server).
Run this before starting the host if you don't know the VM's IP.

Usage:
    python -m utils.vm_discover
"""

import asyncio
import ipaddress
import logging
import os
import socket
import sys
from pathlib import Path

logger = logging.getLogger("vdak.discover")
logging.basicConfig(level=logging.INFO, format="%(message)s")

HOST_IP    = os.getenv("HOST_IP", "192.168.31.42")
VM_PORT    = int(os.getenv("HOST_WS_PORT", "8765"))
TIMEOUT    = 0.5    # seconds per IP probe


def get_subnet(host_ip: str) -> str:
    """Derive /24 subnet from host IP. e.g. 192.168.31.42 → 192.168.31.0/24"""
    network = ipaddress.IPv4Network(f"{host_ip}/24", strict=False)
    return str(network)


def probe_ip(ip: str, port: int, timeout: float) -> bool:
    """Return True if port is open on ip."""
    try:
        with socket.create_connection((ip, port), timeout=timeout):
            return True
    except (socket.timeout, ConnectionRefusedError, OSError):
        return False


async def scan_subnet(subnet: str, port: int) -> list[str]:
    """Scan all IPs in subnet for open port asynchronously."""
    network = ipaddress.IPv4Network(subnet, strict=False)
    loop    = asyncio.get_event_loop()
    tasks   = []

    for host in network.hosts():
        ip = str(host)
        tasks.append(loop.run_in_executor(None, probe_ip, ip, port, TIMEOUT))

    results = await asyncio.gather(*tasks)
    found   = [str(h) for h, ok in zip(network.hosts(), results) if ok]
    return found


def update_env(vm_ip: str):
    """Update VM_WS_URL in vdak-host/.env with discovered IP."""
    env_path = Path(__file__).parent.parent / ".env"
    if not env_path.exists():
        return
    text = env_path.read_text()
    updated = "\n".join(
        f"VM_WS_URL=ws://{vm_ip}:{VM_PORT}" if line.startswith("VM_WS_URL=") else line
        for line in text.splitlines()
    )
    env_path.write_text(updated)
    logger.info(f"✅ .env updated: VM_WS_URL=ws://{vm_ip}:{VM_PORT}")


async def main():
    subnet = get_subnet(HOST_IP)
    logger.info(f"🔍 Scanning {subnet} for VM on port {VM_PORT}...")
    found = await scan_subnet(subnet, VM_PORT)

    if not found:
        logger.warning(
            f"❌ No VM found on port {VM_PORT} in {subnet}.\n"
            f"   Make sure the VM is running and server.py is started."
        )
        sys.exit(1)

    if len(found) == 1:
        vm_ip = found[0]
        logger.info(f"✅ VM found at: {vm_ip}")
        update_env(vm_ip)
    else:
        logger.info(f"⚠️  Multiple hosts found: {found}")
        logger.info("Choosing the first one. Edit .env manually if wrong.")
        vm_ip = found[0]
        update_env(vm_ip)


if __name__ == "__main__":
    asyncio.run(main())
